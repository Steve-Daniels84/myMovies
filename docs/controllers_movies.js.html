require('dotenv').config();
const mongoose = require('mongoose');

// Connect to MongoDB using the connection URI from environment variables
/**
 * Connect to the MongoDB database using the connection string provided in the environment variable `CONNECTION_URI`.
 */
mongoose.connect(process.env.CONNECTION_URI, {
    useNewUrlParser: true,
    useUnifiedTopology: true,
});

const Models = require('../models/models.js');
let Movies = Models.Movie;

/**
 * Retrieves and sends a list of all movies.
 * @function listAllMovies
 * @param {Object} req - The HTTP request object.
 * @param {Object} res - The HTTP response object.
 * @returns {void}
 */
async function listAllMovies(req, res) {
    await Movies.find()
        .then((movies) => {
            res.status(200).json(movies);
        })
        .catch((err) => {
            res.status(400).send('Error: ' + err);
        });
}

/**
 * Retrieves a movie by its ID.
 * @function getMovieById
 * @param {Object} req - The HTTP request object, with `req.params.id` containing the movie ID.
 * @param {Object} res - The HTTP response object.
 * @returns {void}
 */
async function getMovieById(req, res) {
    await Movies.find({ '_id': req.params.id })
        .then((movies) => {
            res.status(200).json(movies);
        })
        .catch((err) => {
            res.status(400).send('Error: ' + err);
        });
}

/**
 * Adds a new movie to the database.
 * @function addMovie
 * @param {Object} req - The HTTP request object, with `req.body` containing the movie details.
 * @param {Object} res - The HTTP response object.
 * @returns {void}
 */
async function addMovie(req, res) {
    await Movies.findOne({ Title: req.body.Title })
        .then((movie) => {
            if (movie) {
                res.status(400).send('Movie already exists');
            } else {
                Movies.create({
                    Genre: {
                        Name: req.body.Genre.Name,
                        Description: req.body.Genre.Description,
                    },
                    Director: {
                        Name: req.body.Director.Name,
                        Bio: req.body.Director.Bio,
                        Birth: req.body.Director.Birth,
                    },
                    Actors: req.body.Actors,
                    Title: req.body.Title,
                    Description: req.body.Description,
                    ImagePath: req.body.ImagePath,
                    Featured: req.body.Featured,
                })
                    .then((movie) => {
                        res.status(201).json(movie);
                    })
                    .catch((error) => {
                        res.status(500).send('Error: ' + error);
                    });
            }
        })
        .catch((error) => {
            res.status(500).send('Error: ' + error);
        });
}

/**
 * Updates the genre of a movie by its title.
 * @function updateGenreByMovieTitle
 * @param {Object} req - The HTTP request object, with `req.params.Title` containing the movie title.
 * @param {Object} res - The HTTP response object, with `req.body` containing the updated genre details.
 * @returns {void}
 */
async function updateGenreByMovieTitle(req, res) {
    try {
        const updatedMovie = await Movies.findOneAndUpdate(
            { Title: req.params.Title },
            {
                $set: {
                    "Genre.Name": req.body.Name,
                    "Genre.Description": req.body.Description,
                },
            },
            { new: true }
        );
        if (updatedMovie) {
            res.status(200).send('Genre details updated');
        } else {
            res.status(404).send('Movie not found');
        }
    } catch (err) {
        res.status(400).send('Error: ' + err);
    }
}

/**
 * Deletes a movie by its ID.
 * @function deleteMovieById
 * @param {Object} req - The HTTP request object, with `req.params.id` containing the movie ID.
 * @param {Object} res - The HTTP response object.
 * @returns {void}
 */
async function deleteMovieById(req, res) {
    const movieId = req.params.id;
    await Movies.findOneAndDelete({ _id: req.params.id })
        .then((movies) => {
            if (!movies) {
                res.status(400).send('Movie ' + movieId + ' was not found');
            } else {
                res.status(200).send('Movie ' + movieId + ' was deleted');
            }
        })
        .catch((err) => {
            console.error(err);
            res.status(500).send('Error: ' + err);
        });
}
