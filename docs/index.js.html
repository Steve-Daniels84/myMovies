require('dotenv').config();

// Access modules
const express = require('express'), 
    morgan = require('morgan'),
    fs = require('fs'),
    path = require('path'),
    bodyParser = require('body-parser'),
    mongoose = require('mongoose'),
    movies = require('./controllers/movies.js'),
    users = require('./controllers/users.js'),
    cors = require('cors');

const { check, validationResult } = require('express-validator');
const { requiredRole, validateInput, validateUserId } = require('./validation/validations.js');
const { createUserRules, updateUserRules } = require('./validation/rules.js');

const app = express();

app.use(cors());
app.use(bodyParser.json());

let auth = require('./controllers/auth/auth.js')(app);

const passport = require('passport');
require('./controllers/auth/passport.js');

/**
 * Create a write stream for logging HTTP requests to a log file.
 */
const accessLogStream = fs.createWriteStream(path.join(__dirname, 'log.txt'), { flags: 'a' }); 

/**
 * Use Morgan middleware for logging HTTP requests.
 */
app.use(morgan('combined', { stream: accessLogStream })); 

/**
 * Serve static files from the "public" directory.
 */
app.use(express.static('public'));

/**
 * Error handling middleware.
 * Logs errors to the console and sends a generic 500 error message.
 */
app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).send('Internal server error');
});

/**
 * Route to the documentation page as the default endpoint.
 */
app.get('/', (req, res) => {
    res.status(200).sendFile('documentation.html', { root: __dirname + '/public' });
});

/**
 * Route to the documentation page.
 */
app.get('/documentation', (req, res) => {
    res.status(200).sendFile('documentation.html', { root: __dirname + '/public' });
});

/**
 * Movie routes.
 * These routes manage movie-related operations such as listing, retrieving, adding, updating, and deleting movies.
 */
app.get('/movies', requiredRole('user'), passport.authenticate('jwt', { session: false }), movies.listAllMovies); // List all movies
app.get('/movies/id/:id', requiredRole('user'), passport.authenticate('jwt', { session: false }), movies.getMovieById); // Get a movie by ID
app.get('/movies/:Genre', requiredRole('user'), passport.authenticate('jwt', { session: false }), movies.getMoviesByGenre); // Get movies by genre
app.get('/movies/:Title', requiredRole('user'), passport.authenticate('jwt', { session: false }), movies.getMovieByTitle); // Get a movie by title
app.post('/movies', requiredRole('sysAdmin'), passport.authenticate('jwt', { session: false }), movies.addMovie); // Add a new movie
app.put('/movies/:Title', requiredRole('sysAdmin'), passport.authenticate('jwt', { session: false }), movies.updateGenreByMovieTitle); // Update genre info for a movie by its title
app.delete('/movies/:id', requiredRole('sysAdmin'), passport.authenticate('jwt', { session: false }), movies.deleteMovieById); // Delete a movie by ID

/**
 * User routes.
 * These routes manage user-related operations such as listing, retrieving, adding, updating, and deleting users.
 */
app.get('/users', requiredRole('sysAdmin'), passport.authenticate('jwt', { session: false }), users.listUsers); // List all users
app.get('/users/:id', requiredRole('user'), passport.authenticate('jwt', { session: false }), validateUserId, users.getUserById); // Get a user by ID
app.put('/users/:id', updateUserRules, validateInput, requiredRole('user'), passport.authenticate('jwt', { session: false }), validateUserId, users.updateUser); // Update user details
app.post('/users/', createUserRules, validateInput, users.addUser); // Add a new user
app.put('/users/:id/:movieId', requiredRole('user'), passport.authenticate('jwt', { session: false }), validateUserId, users.addMovie); // Add a movie to a user's favorites
app.delete('/users/:id', requiredRole('user'), passport.authenticate('jwt', { session: false }), validateUserId, users.deleteUser); // Delete a user
app.delete('/users/:id/:movieId', requiredRole('user'), passport.authenticate('jwt', { session: false }), validateUserId, users.deleteMovie); // Remove a movie from a user's favorites

/**
 * Start the server.
 * Listens on the specified port (default: 8080).
 */
const port = process.env.PORT || 8080;
app.listen(port, '0.0.0.0', () => {
    console.log('Listening on port: ' + port);
});
